# Mutations of glam's code, each a fault a law of glam/LAWS.bend named
# should catch (scripts/one_law.py; docs/Proving.md, What to claim):
#   python3 -I scripts/one_law.py glam/PROOF.bend LAW --mutants glam/MUTANTS.txt
#
# Unlinking a cursor from its ring (unlink_keeps): each mutation writes a
# word the law says is left alone.

=== unlink_word2 unlink_keeps unlink_closed
the predecessor's word 2, its position, given the cursor's next
--- ring.bend
  P.wr(h, (p + 3 : U32), nx)
+++
  P.wr(h, (p + 2 : U32), nx)

=== unlink_clears_own unlink_keeps
the cursor unlinked has its own word 3 cleared too
--- ring.bend
  rg.unlink.set(rg.next(h, x), p)
+++
  P.wr(rg.unlink.set(rg.next(h, x), p), (x + 3 : U32), 0)

# The ring's walk (pred_in_ring): each mutation lets the node found leave
# the ring.

=== pred_next_node pred_in_ring
the node found reported as the one after it in the heap
--- ring.bend
    case Rw{h, +n, +p, found, done}:
      (h, p)
+++
    case Rw{h, +n, +p, found, done}:
      (h, (p + 4 : U32))

=== walk_by_position pred_in_ring
the walk follows each cursor's word 2, its position, not its word 3
--- ring.bend
def rg.next(h: P.H, +x: U32) -> P.H & U32:
  P.rd(h, (x + 3 : U32))
+++
def rg.next(h: P.H, +x: U32) -> P.H & U32:
  P.rd(h, (x + 2 : U32))

# The ring closed (unlink_closed, insert_closed): each mutation leaves a
# link out of the ring's members.

=== unlink_to_self unlink_closed
the predecessor given the cursor unlinked, not the cursor's next
--- ring.bend
  rg.unlink.set(rg.next(h, x), p)
+++
  rg.unlink.set((h, x), p)

=== insert_word2 insert_closed
the cursor inserted given its next in its word 2, its position
--- ring.bend
  P.wr(P.wr(h, (x + 3 : U32), n), (situ + 3 : U32), x)
+++
  P.wr(P.wr(h, (x + 2 : U32), n), (situ + 3 : U32), x)

=== insert_next_by_position insert_closed
the cursor inserted given situ's word 2, its position, as its next
--- ring.bend
  rg.insert.r(rg.next(h, situ), situ, x)
+++
  rg.insert.r(P.rd(h, (situ + 2 : U32)), situ, x)

# A join (join_keeps): each mutation writes a word outside the ring's
# members, the free list's head and the faced ports' link words.

=== join_unlink_faced join_keeps
the port x faced unlinked, not x
--- ring.bend
  P.free(P.free(rg.unlink(rg.unlink(P.link_xx(h, u, w), x), y), x), y)
+++
  P.free(P.free(rg.unlink(rg.unlink(P.link_xx(h, u, w), u), y), x), y)

=== join_link_next_port join_keeps
the port y faced linked one word on
--- ring.bend
  P.free(P.free(rg.unlink(rg.unlink(P.link_xx(h, u, w), x), y), x), y)
+++
  P.free(P.free(rg.unlink(rg.unlink(P.link_xx(h, u, (w + 1 : U32)), x), y), x), y)

=== join_frees_next_node join_keeps
the node after x freed, not x
--- ring.bend
  P.free(P.free(rg.unlink(rg.unlink(P.link_xx(h, u, w), x), y), x), y)
+++
  P.free(P.free(rg.unlink(rg.unlink(P.link_xx(h, u, w), x), y), (x + 4 : U32)), y)

=== join_port_by_position join_keeps
y's word 2, its position, read as the port it faced
--- ring.bend
  jn.w(P.rd(h, (y + 1 : U32)), u, x, y)
+++
  jn.w(P.rd(h, (y + 2 : U32)), u, x, y)
