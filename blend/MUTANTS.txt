# Mutations of Blend's code, each a fault a law named should catch
# (scripts/one_law.py; docs/Proving.md, What to claim). Try them after a
# proof or the code it follows changes:
#   python3 -I scripts/one_law.py blend/PROOF.bend LAW --mutants blend/MUTANTS.txt

=== root_kept order.ends
a root already walked stays among the roots: the walk loops
--- blend.bend
    case True{}:
      Ds{Nil{}, rt, done, order, ers}
+++
    case True{}:
      Ds{Nil{}, Con{r, rt}, done, order, ers}

=== call_kept order.ends
a call to a unit walked is not consumed: the walk loops
--- blend.bend
    case True{} _:
      Ds{Con{Fm{u, rs}, st}, roots, done, order, ers}
+++
    case True{} _:
      Ds{Con{Fm{u, Con{c, rs}}, st}, roots, done, order, ers}

=== done_forgets order.ends
a unit popped is not marked walked, so may be walked again
--- blend.bend
    case True{}:
      Ds{st, roots, Con{u, done}, Con{u, order}, ers}
+++
    case True{}:
      Ds{st, roots, done, Con{u, order}, ers}

=== fuel_short order.ends
a fuel of the roots alone
--- blend.bend
  Nat.add(walk.turns(roots), walk.calls(cal))
+++
  walk.turns(roots)

=== turns_one order.ends
one turn for each root and call, where a root with no calls takes two
--- blend.bend
      2n+walk.turns(t)
+++
      1n+walk.turns(t)

=== fast_seen_order order.fast
a root found walked by the order's set, which a unit held back is not in
--- blend.bend
              df.root.k(vs.has(vd, r), r, rt, done, order, ers, vd, vo, ve, ct)
+++
              df.root.k(vs.has(vo, r), r, rt, done, order, ers, vd, vo, ve, ct)

=== fast_unentered order.fast
a unit pushed and not entered, so that a cycle through it goes unseen

      Df{Con{Fm{c, cs.at(ct, c)}, Con{Fm{u, rs}, st}}, roots, done, order, ers, vd, vo, vs.put(ve, c)}
+++
      Df{Con{Fm{c, cs.at(ct, c)}, Con{Fm{u, rs}, st}}, roots, done, order, ers, vd, vo, ve}

=== fast_unwalked order.fast
a unit popped into the order but not into the set of those walked

+++
      Df{st, roots, Con{u, done}, Con{u, order}, ers, vd, vs.put(vo, u), ve}

=== fast_calls_swapped order.fast

                case PwNode{e, o}:

=== copies_from_two emit.defs
a unit's numbered copies written from the second on, the first missing

=== names_shifted emit.names
the names check counting a unit's names from its first copy, not its own
--- blend.bend
      List.append(&2, Nm, names.copies(1n+U32.to_nat(mu.m(ec.mu(e, x))), e, x, 0), names.all(t, e))
+++
      List.append(&2, Nm, names.copies(1n+U32.to_nat(mu.m(ec.mu(e, x))), e, x, 1), names.all(t, e))

=== hand_reversed mults.uses
a user's uses handed on the first first, so that each callee's are in
the reverse of the user's order
--- blend.bend
          pend.cons(pend.uses(t, u, p), c, Ru{u, no})
+++
          pend.uses(t, u, pend.cons(p, c, Ru{u, no}))

=== hand_forgets mults.uses
a use handed on replacing the uses waiting, not joining them
--- blend.bend
  um.put(List<&2, Ru>, p, c, Con{r, pend.at(p, c)})
+++
  um.put(List<&2, Ru>, p, c, [r])
